CREATE TABLE O_RollCall (
  roll_no NUMBER PRIMARY KEY,
  name    VARCHAR2(30)
);

CREATE TABLE N_RollCall (
  roll_no NUMBER,
  name    VARCHAR2(30)
);

BEGIN
  INSERT INTO O_RollCall VALUES (1, 'Amit');
  INSERT INTO O_RollCall VALUES (2, 'Riya');
  INSERT INTO O_RollCall VALUES (3, 'Sohan');
  COMMIT;
END;
/

BEGIN
  INSERT INTO N_RollCall VALUES (2, 'Riya');    -- duplicate (should be skipped)
  INSERT INTO N_RollCall VALUES (3, 'Sohan');   -- duplicate (should be skipped)
  INSERT INTO N_RollCall VALUES (4, 'Meena');   -- new (should be inserted)
  INSERT INTO N_RollCall VALUES (5, 'Raj');     -- new (should be inserted)
  COMMIT;
END;
/

DECLARE
  v_exists NUMBER;

  CURSOR c_new(p_roll N_RollCall.roll_no%TYPE) IS
    SELECT roll_no, name
    FROM N_RollCall
    WHERE roll_no = p_roll;

BEGIN
  FOR rec IN (SELECT roll_no FROM N_RollCall) LOOP
    SELECT COUNT(*)
    INTO v_exists
    FROM O_RollCall
    WHERE roll_no = rec.roll_no;

    IF v_exists = 0 THEN
      FOR rec_new IN c_new(rec.roll_no) LOOP
        INSERT INTO O_RollCall (roll_no, name)
        VALUES (rec_new.roll_no, rec_new.name);
      END LOOP;
    END IF;
  END LOOP;

  COMMIT;
END;
/

SELECT * FROM O_RollCall ORDER BY roll_no;
